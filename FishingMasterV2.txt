local _k = {173,59,201,88,142}

local function _r(t)
    local n, o = #t, {}
    for i = 1, n do
        local v = t[n - i + 1]
        v = (v - i * 7) % 256
        v = bit32.bxor(v, _k[(i - 1) % #_k + 1])
        o[i] = string.char(v)
    end
    return table.concat(o)
end

-- Script cũ - thông báo và chạy script mới
local oldScript = "https://raw.githubusercontent.com/KimLong-Valencia/REAL/refs/heads/main/FishinMasterV3.txt"

-- Thông báo cho người dùng
local StarterGui = game:GetService("StarterGui")
StarterGui:SetCore("SendNotification", {
    Title = "Script đã cũ",
    Text = "Script này đã lỗi thời. Đang tải script mới...",
    Duration = 5
})

-- Chờ một chút để thông báo hiển thị
task.wait(2)

-- Chạy script mới
local success, err = pcall(function()
    loadstring(game:HttpGet(oldScript))()
end)

if not success then
    StarterGui:SetCore("SendNotification", {
        Title = "Lỗi",
        Text = "Không thể tải script mới: " .. tostring(err),
        Duration = 5
    })
end
